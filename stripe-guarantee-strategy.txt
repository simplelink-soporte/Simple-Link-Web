   interface ValidationContext {
     booking: BookingDetails;
     stripeAccount: StripeAccountInfo;
     customer: CustomerInfo;

   class ValidationService {
     async validateCharge(context: ValidationContext): Promise<ValidationResult> {
       // 1. Validar booking
       await this.validateBooking(context.booking);
       
       // 2. Validar cuenta Stripe
       await this.validateStripeAccount(context.stripeAccount);
       
       // 3. Validar customer
       await this.validateCustomer(context.customer);
       
       // 4. Validar límites y restricciones
       await this.validateLimits(context);
     }
   }
