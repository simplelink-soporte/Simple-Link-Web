```sql
BEGIN
    -- Verificar si los tipos existen
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'payment_type') THEN
        CREATE TYPE payment_type AS ENUM ('booking', 'guarantee', 'no_show_charge');
    ELSE
        -- Agregar nuevos valores al enum existente
        ALTER TYPE payment_type ADD VALUE IF NOT EXISTS 'guarantee';
        ALTER TYPE payment_type ADD VALUE IF NOT EXISTS 'no_show_charge';
    END IF;
END $$;

-- 2. Modificar tabla payments
ALTER TABLE payments
    ADD COLUMN IF NOT EXISTS stripe_payment_method_id text,
    ADD COLUMN IF NOT EXISTS stripe_account_id text,
    ADD COLUMN IF NOT EXISTS parent_payment_id uuid REFERENCES payments(id),
    ADD COLUMN IF NOT EXISTS charge_reason text;

-- 3. Crear índices para optimización
CREATE INDEX IF NOT EXISTS idx_payments_stripe_account ON payments(stripe_account_id);
CREATE INDEX IF NOT EXISTS idx_payments_parent_payment ON payments(parent_payment_id);
CREATE INDEX IF NOT EXISTS idx_payments_type_status ON payments(payment_type, payment_status);

-- 4. Comentarios para documentación
COMMENT ON COLUMN payments.stripe_payment_method_id IS 'ID del método de pago en Stripe';
COMMENT ON COLUMN payments.stripe_account_id IS 'ID de la cuenta Connect del club';
COMMENT ON COLUMN payments.parent_payment_id IS 'ID del pago de garantía original para cargos no-show';
COMMENT ON COLUMN payments.charge_reason IS 'Razón del cargo (ej: no-show)';
```

3. **Proceso de Cargo**
   ```typescript
   interface NoShowChargeParams {
     bookingId: string;
     reason: string;
     shouldCharge: boolean;
   }

   async function handleNoShow({
     bookingId,
     reason,
     shouldCharge
   }: NoShowChargeParams) {
     // 1. Obtener datos necesarios
     const [booking, guarantee] = await Promise.all([
       getBookingDetails(bookingId),
       getGuaranteePayment(bookingId)
     ]);

     // 2. Validaciones
     if (shouldCharge) {
       await validateChargeEligibility(booking, guarantee);
     }

     // 3. Proceso de cargo
     if (shouldCharge) {
       const chargeAmount = calculateNoShowCharge(booking.total_price);
       
       // Crear Payment Intent
       const paymentIntent = await stripe.paymentIntents.create({
         amount: Math.round(chargeAmount * 100),
         currency: 'eur',
         payment_method: guarantee.stripe_payment_method_id,
         customer: guarantee.stripe_customer_id,
         off_session: true,
         confirm: true,
         metadata: {
           booking_id: bookingId,
           charge_type: 'no_show',
           original_guarantee_id: guarantee.id
         }
       }, {
         stripeAccount: guarantee.stripe_account_id
       });

       // Registrar el cargo en la base de datos
       await createNoShowPaymentRecord({
         bookingId,
         paymentIntent,
         guarantee,
         chargeAmount,
         reason
       });
     }

     // 4. Actualizar estado de la reserva
     await updateBookingStatus(bookingId, {
       cancelled: true,
       cancellationReason: reason,
       chargeApplied: shouldCharge
     });

     // 5. Enviar notificaciones
     await sendNotifications({
       booking,
       chargeApplied: shouldCharge,
       chargeAmount
     });
   }
   ```

4. **Manejo de Errores**
   ```typescript
   async function validateChargeEligibility(booking: Booking, guarantee: Payment) {
     const errors: string[] = [];

     // Validar estado de la reserva
     if (booking.cancelled_at) {
       errors.push('La reserva ya está cancelada');
     }

     // Validar garantía
     if (!guarantee || guarantee.payment_type !== 'guarantee') {
       errors.push('No existe una garantía válida para esta reserva');
     }

     // Validar método de pago
     if (guarantee && !guarantee.stripe_payment_method_id) {
       errors.push('No hay un método de pago válido asociado');
     }

     // Validar que no hay un cargo previo
     const existingCharge = await getExistingNoShowCharge(booking.id);
     if (existingCharge) {
       errors.push('Ya existe un cargo por no-show para esta reserva');
     }

     if (errors.length > 0) {
       throw new ValidationError(errors.join('. '));
     }
   }
   ```

5. **Notificaciones y Registro**
   ```typescript
   async function sendNotifications({
     booking,
     chargeApplied,
     chargeAmount
   }: NotificationParams) {
     // Email al cliente
     await sendCustomerEmail({
       type: chargeApplied ? 'no_show_charge' : 'cancellation',
       booking,
       chargeAmount
     });

     // Notificación al club
     await sendClubNotification({
       type: 'booking_cancelled',
       booking,
       chargeApplied,
       chargeAmount
     });

     // Registro en el sistema
     await logBookingEvent({
       type: chargeApplied ? 'no_show_charge' : 'cancellation',
       bookingId: booking.id,
       metadata: {
         chargeApplied,
         chargeAmount
       }
     });
   }
   ```

### 3.1 Flujo de Captura de Tarjeta

1. **Trigger**: Usuario selecciona "Garantía" como tipo de pago
2. **Proceso**:
   ```typescript
   async function setupGuarantee(bookingId: string) {
     // 1. Verificar cuenta Connect del club
     const { data: club } = await supabase
       .from('stripe_connections')
       .select('stripe_account_id')
       .single();

     // 2. Crear Setup Intent
     const setupIntent = await stripe.setupIntents.create({
       payment_method_types: ['card'],
       usage: 'off_session',
     }, {
       stripeAccount: club.stripe_account_id
     });

     // 3. Registrar en payments
     await supabase.from('payments').insert({
       booking_id: bookingId,
       payment_type: 'guarantee',
       payment_method: 'stripe',
       stripe_account_id: club.stripe_account_id,
       payment_status: 'pending'
     });

     return setupIntent.client_secret;
   }
   ```

### 3.2 Flujo de Cargo por No-Show

1. **Trigger**: Club marca reserva como no-show
2. **Proceso**:
   ```typescript
   async function chargeNoShow(bookingId: string) {
     // 1. Obtener garantía original
     const { data: guarantee } = await supabase
       .from('payments')
       .select('*')
       .eq('booking_id', bookingId)
       .eq('payment_type', 'guarantee')
       .single();

     // 2. Calcular monto (30%)
     const { data: booking } = await supabase
       .from('bookings')
       .select('total_price')
       .eq('id', bookingId)
       .single();

     const chargeAmount = booking.total_price * 0.3;

     // 3. Crear cargo
     const paymentIntent = await stripe.paymentIntents.create({
       amount: Math.round(chargeAmount * 100),
       currency: 'eur',
       payment_method: guarantee.stripe_payment_method_id,
       off_session: true,
       confirm: true
     }, {
       stripeAccount: guarantee.stripe_account_id
     });

     // 4. Registrar cargo
     await supabase.from('payments').insert({
       booking_id: bookingId,
       payment_type: 'no_show_charge',
       payment_method: 'stripe',
       stripe_account_id: guarantee.stripe_account_id,
       total_price: chargeAmount,
       parent_payment_id: guarantee.id,
       payment_status: 'succeeded'
     });
   }
   ```

### 4.2 Nuevo Componente: GuaranteeCardForm
```typescript
interface GuaranteeCardFormProps {
  stripeAccountId: string;
  onSuccess: (paymentMethodId: string) => void;
  onError: (error: Error) => void;
}

export function GuaranteeCardForm({
  stripeAccountId,
  onSuccess,
  onError
}: GuaranteeCardFormProps) {
  const stripe = useStripe();
  const elements = useElements();
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (event: React.FormEvent) => {
    event.preventDefault();
    if (!stripe || !elements) return;

    setLoading(true);
    setError(null);

    try {
      // 1. Crear Setup Intent
      const response = await fetch('/api/stripe/setup-intent', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ stripeAccountId })
      });

      const { clientSecret } = await response.json();

      // 2. Confirmar Setup Intent
      const result = await stripe.confirmCardSetup(clientSecret, {
        payment_method: {
          card: elements.getElement(CardElement)!,
        }
      });

      if (result.error) {
        setError(result.error.message || 'Error al procesar la tarjeta');
        onError(result.error);
      } else {
        onSuccess(result.setupIntent.payment_method as string);
      }
    } catch (err) {
      setError('Error al procesar la tarjeta');
      onError(err as Error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      <div className="p-4 border rounded-lg">
        <CardElement
          options={{
            style: {
              base: {
                fontSize: '16px',
                color: '#424770',
                '::placeholder': {
                  color: '#aab7c4',
                },
              },
              invalid: {
                color: '#9e2146',
              },
            },
          }}
        />
      </div>
      
      {error && (
        <div className="text-sm text-red-600">
          {error}
        </div>
      )}

      <Button
        type="submit"
        disabled={!stripe || loading}
        className="w-full"
      >
        {loading ? 'Procesando...' : 'Guardar Tarjeta'}
      </Button>
    </form>
  );
}
```
