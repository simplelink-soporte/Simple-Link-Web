create table
  public.empresas (
    id uuid not null default gen_random_uuid (),
    name text not null,
    business_name text null,
    email text null,
    phone text null,
    address text null,
    city text null,
    state text null,
    is_active boolean null default true,
    settings jsonb null default '{}'::jsonb,
    created_at timestamp with time zone null default timezone ('utc'::text, now()),
    updated_at timestamp with time zone null default timezone ('utc'::text, now()),
    auth_user_id uuid null,
    plan_type public.plans_types_enums null default 'Free'::plans_types_enums,
    onboarding public.onboarding_enum null,
    country text null,
    constraint organizations_pkey primary key (id),
    constraint empresas_user_id_key unique (auth_user_id),
    constraint empresas_auth_user_id_fkey foreign key (auth_user_id) references auth.users (id) deferrable initially deferred
  ) tablespace pg_default;

create index if not exists idx_empresas_email on public.empresas using btree (email) tablespace pg_default;

create index if not exists idx_empresas_is_active on public.empresas using btree (is_active) tablespace pg_default;

create index if not exists idx_empresas_user_id on public.empresas using btree (auth_user_id) tablespace pg_default;

create trigger handle_new_adminprofile_trigger
after insert
or
update of business_name on empresas for each row
execute function handle_new_adminprofile ();
