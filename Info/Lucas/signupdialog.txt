"use client"

import { Button } from "@/components/ui/Button"
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from "@/components/ui/dialog"
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { cn } from "@/lib/utils"
import PhoneInput from "@/components/ui/inputphone"
import { useState, useEffect } from "react"
import PasswordInput from "@/components/ui/password"
import { Loader2 } from "lucide-react"
import { toast } from "sonner"
import { clienteService, ClienteServiceError } from "@/services/clienteService"
import { validateEmail, validatePassword, validatePhone } from "@/utils/validation"
import { ClienteRegistro } from "@/types"
import { MotionDiv } from "@/components/ui/motion"
import { supabase } from "@/lib/supabase"

interface SignUpDialogProps {
  className?: string
  size?: "default" | "sm" | "lg"
  children?: React.ReactNode
  open?: boolean
  onOpenChange?: (open: boolean) => void
}

export function SignUpDialog({ className, size = "default", children, open: controlledOpen, onOpenChange }: SignUpDialogProps) {
  const [isOpen, setIsOpen] = useState(controlledOpen || false)
  const [formData, setFormData] = useState({
    phoneNumber: "",
    password: "",
    email: "",
    name: ""
  })
  const [isLoading, setIsLoading] = useState(false)
  const [isVerificationSent, setIsVerificationSent] = useState(false)

  // Sincronizar estado controlado
  useEffect(() => {
    if (controlledOpen !== undefined) {
      setIsOpen(controlledOpen)
    }
  }, [controlledOpen])

  const handleClose = () => {
    setIsOpen(false)
    onOpenChange?.(false)
    resetForm()
  }

  const initialFormState = {
    phoneNumber: "",
    password: "",
    email: "",
    name: ""
  }

  const resetForm = () => {
    setFormData(initialFormState)
  }

  const handleInputChange = (field: string, value: string) => {
    setFormData(prev => ({ ...prev, [field]: value }))
  }

  const validateForm = (): string | null => {
    const { email, password, name, phoneNumber } = formData

    if (!email || !password || !name) {
      return "Por favor completa todos los campos requeridos"
    }

    const cleanEmail = email.trim().toLowerCase()
    
    if (!validateEmail(cleanEmail)) {
      return "Por favor ingresa un email válido (ejemplo: usuario@dominio.com)"
    }

    // Validar requisitos de contraseña
    const passwordRequirements = [
      { regex: /.{8,}/, message: "La contraseña debe tener al menos 8 caracteres" },
      { regex: /[0-9]/, message: "La contraseña debe contener al menos 1 número" },
      { regex: /[a-z]/, message: "La contraseña debe contener al menos 1 letra minúscula" },
      { regex: /[A-Z]/, message: "La contraseña debe contener al menos 1 letra mayúscula" },
      { regex: /[!@#$%^&*(),.?":{}|<>]/, message: "La contraseña debe contener al menos 1 caracter especial (!@#$%^&*)" },
    ]

    for (const requirement of passwordRequirements) {
      if (!requirement.regex.test(password)) {
        return requirement.message
      }
    }

    if (phoneNumber && !validatePhone(phoneNumber)) {
      return "Por favor ingresa un número de teléfono válido (solo números)"
    }

    return null
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    
    const validationError = validateForm()
    if (validationError) {
      toast.error(validationError)
      return
    }
    
    try {
      setIsLoading(true)
      
      const { data, error } = await clienteService.registrarCliente({
        email: formData.email,
        password: formData.password,
        name: formData.name,
        phone: formData.phoneNumber || undefined,
        role: 'empresa'
      })

      if (error) throw error

      // Mostrar diálogo de verificación
      setIsVerificationSent(true)

      // Éxito - mostrar mensaje toast con estilo mejorado
      toast.success(
        "¡Registro exitoso!",
        { 
          duration: 4000,
          className: "text-center font-medium",
          description: "Por favor revisa tu email para verificar tu cuenta.",
          position: "top-center"
        }
      )
      
      // Limpiar formulario y cerrar diálogo de registro
      setFormData(initialFormState)
      setIsVerificationSent(false)

    } catch (error) {
      console.error('Error en registro:', error)
      if (error instanceof ClienteServiceError) {
        toast.error(error.message)
      } else {
        toast.error('Error inesperado. Por favor intenta nuevamente.')
      }
    } finally {
      setIsLoading(false)
    }
  }

  const handleGoogleSignUp = async () => {
    try {
      setIsLoading(true)
      const { data, error } = await supabase.auth.signInWithOAuth({
        provider: 'google',
        options: {
          redirectTo: `${window.location.origin}/auth/callback`,
          queryParams: {
            prompt: 'select_account',
            access_type: 'offline'
          }
        }
      })

      if (error) throw error

    } catch (error) {
      console.error('Error al registrarse con Google:', error)
      toast.error('No se pudo completar el registro con Google. Por favor intenta nuevamente.')
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <>
      <Dialog 
        open={isOpen} 
        onOpenChange={(open) => {
          setIsOpen(open)
          onOpenChange?.(open)
          if (!open) {
            resetForm()
            setIsVerificationSent(false)
            history.pushState(null, "", window.location.href)
          }
        }}
      >
        <DialogTrigger asChild>
          <Button 
            size={size}
            className={cn(
              "w-full min-w-[160px] rounded-full font-medium sm:w-auto",
              className
            )}
          >
            {children || "¡Empezar Gratis!"}
          </Button>
        </DialogTrigger>
        <DialogContent 
          className="sm:max-w-[425px] p-0 gap-0 border-none bg-background"
          onClose={handleClose}
          title="Crear cuenta"
        >
          <div className="p-6">
            {/* Indicador de arrastre para móvil */}
            <div className="w-12 h-1 bg-border/60 rounded-full mx-auto mb-6 sm:hidden" />
            
            <MotionDiv
              className="flex flex-col items-center gap-2"
              initial={{ opacity: 0, y: -20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ duration: 0.3 }}
            >
              <DialogHeader>
                <DialogTitle className="sm:text-center">Crear una cuenta</DialogTitle>
                <DialogDescription className="sm:text-center">
                  Completa los siguientes campos para comenzar.
                </DialogDescription>
              </DialogHeader>
            </MotionDiv>

            <MotionDiv 
              className="space-y-5"
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              transition={{ duration: 0.3, delay: 0.1 }}
            >
              <form onSubmit={handleSubmit} className="space-y-4">
                <div className="space-y-2">
                  <Label htmlFor="signup-name">Nombre completo</Label>
                  <Input 
                    id="signup-name" 
                    placeholder="Juan Pérez" 
                    type="text" 
                    required 
                    value={formData.name}
                    onChange={(e) => handleInputChange('name', e.target.value)}
                  />
                </div>
                <div className="space-y-2">
                  <Label htmlFor="signup-email">Email</Label>
                  <Input 
                    id="signup-email" 
                    placeholder="juan@ejemplo.com" 
                    type="email" 
                    required
                    value={formData.email}
                    onChange={(e) => handleInputChange('email', e.target.value)}
                  />
                </div>
                <div className="space-y-2">
                  <Label htmlFor="signup-phone">Teléfono</Label>
                  <PhoneInput
                    value={formData.phoneNumber}
                    onChange={(value) => handleInputChange('phoneNumber', value)}
                    id="signup-phone"
                    required
                    placeholder="Ingresa tu número de teléfono"
                  />
                  <p className="text-xs text-muted-foreground">
                    Solo números, sin espacios ni caracteres especiales
                  </p>
                </div>
                <div className="space-y-2">
                  <Label htmlFor="signup-password">Contraseña</Label>
                  <PasswordInput
                    id="signup-password"
                    value={formData.password}
                    onChange={(value) => handleInputChange('password', value)}
                    required
                    placeholder="Ingresa tu contraseña"
                    disabled={isLoading}
                    showRequirements={true}
                  />
                </div>
                <Button type="submit" className="w-full" disabled={isLoading}>
                  {isLoading ? (
                    <Loader2 className="h-4 w-4 animate-spin" />
                  ) : (
                    "Crear cuenta"
                  )}
                </Button>
              </form>
            </MotionDiv>

            <MotionDiv 
              className="flex items-center gap-3 before:h-px before:flex-1 before:bg-border after:h-px after:flex-1 after:bg-border"
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              transition={{ duration: 0.3, delay: 0.2 }}
            >
              <span className="text-xs text-muted-foreground">O</span>
            </MotionDiv>

            <MotionDiv
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              transition={{ duration: 0.3, delay: 0.3 }}
            >
              <Button 
                variant="outline" 
                className="w-full"
                onClick={handleGoogleSignUp}
                disabled={isLoading}
              >
                Continuar con Google
              </Button>
            </MotionDiv>

            <MotionDiv 
              className="text-center text-xs text-muted-foreground"
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              transition={{ duration: 0.3, delay: 0.4 }}
            >
              Al registrarte aceptas nuestros{" "}
              <a className="underline hover:no-underline" href="#">
                Términos y condiciones
              </a>
              .
            </MotionDiv>
          </div>
        </DialogContent>
      </Dialog>

      {isVerificationSent && (
        <Dialog 
          open={isVerificationSent} 
          onOpenChange={(open) => {
            setIsVerificationSent(open)
            if (!open) {
              history.pushState(null, "", window.location.href)
            }
          }}
        >
          <DialogContent 
            className="sm:max-w-[425px]" 
            hideClose
            onClose={() => setIsVerificationSent(false)}
            title="Verificación enviada"
          >
            {/* Indicador de arrastre para móvil */}
            <div className="w-12 h-1 bg-border/60 rounded-full mx-auto mb-6 sm:hidden" />
            
            <MotionDiv
              className="flex flex-col items-center gap-4 py-4"
              initial={{ opacity: 0, y: -20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ duration: 0.3 }}
            >
              <DialogHeader className="text-center">
                <DialogTitle className="text-2xl font-semibold text-center">¡Casi listo!</DialogTitle>
                <DialogDescription className="mt-2 text-center">
                  Te hemos enviado un correo de verificación a <span className="font-medium text-primary">{formData.email}</span>.
                  <br /><br />
                  Por favor revisa tu bandeja de entrada y sigue las instrucciones para activar tu cuenta.
                </DialogDescription>
              </DialogHeader>
              
              <Button 
                onClick={() => setIsVerificationSent(false)}
                className="mt-2 min-w-[120px]"
              >
                Entendido
              </Button>
            </MotionDiv>
          </DialogContent>
        </Dialog>
      )}
    </>
  )
}
