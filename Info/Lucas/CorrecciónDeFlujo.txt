# Plan de Corrección de Flujo - Simple-link

## 1. Cambios en Supabase (Pendientes de Aplicar)

### 1.1 Estructura de Roles y Metadatos
```sql
-- Actualizar app_metadata para todos los usuarios admin
UPDATE auth.users
SET raw_app_meta_data = jsonb_set(
  raw_app_meta_data,
  '{role}',
  '"admin"'
)
WHERE id IN (SELECT auth_user_id FROM public.empresas);

-- Actualizar app_metadata para todos los usuarios cliente
UPDATE auth.users
SET raw_app_meta_data = jsonb_set(
  raw_app_meta_data,
  '{role}',
  '"client"'
)
WHERE id NOT IN (SELECT auth_user_id FROM public.empresas);
```

### 1.2 Políticas de Seguridad
```sql
-- Política para empresas (solo admins)
CREATE POLICY "Solo admins pueden acceder a empresas"
ON public.empresas
FOR ALL
USING (auth.jwt() ->> 'role' = 'admin');

-- Política para vinculaciones (clientes y admins de su empresa)
CREATE POLICY "Acceso a vinculaciones"
ON public.vinculaciones
FOR ALL
USING (
  (auth.jwt() ->> 'role' = 'admin' AND empresa_id IN (
    SELECT id FROM public.empresas WHERE auth_user_id = auth.uid()
  ))
  OR
  (auth.jwt() ->> 'role' = 'client' AND user_id = auth.uid())
);
```

#### 2.3.3 Middleware
```typescript
// src/middleware/adminAuth.ts
export const adminAuthMiddleware = async (req: NextRequest) => {
  const supabase = createServerSupabaseClient(req);
  const session = await supabase.auth.getSession();
  
  if (!session || session.user?.app_metadata?.role !== 'admin') {
    return NextResponse.redirect('/admin/login');
  }
}

// src/middleware/clientAuth.ts
export const clientAuthMiddleware = async (req: NextRequest) => {
  const supabase = createServerSupabaseClient(req);
  const session = await supabase.auth.getSession();
  
  if (!session || session.user?.app_metadata?.role !== 'client') {
    return NextResponse.redirect('/clases/login');
  }
}
```
