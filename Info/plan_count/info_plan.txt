Primer SQL Ejecutado para aplicar el Plan:

-- Create ENUM types
CREATE TYPE plan_type AS ENUM ('FREE', 'PRO');
CREATE TYPE reset_period AS ENUM ('DAILY', 'WEEKLY', 'MONTHLY');

-- Create subscription_plans table
CREATE TABLE subscription_plans (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL UNIQUE,
    type plan_type NOT NULL,
    price DECIMAL(10,2) NOT NULL DEFAULT 0,
    daily_booking_limit INTEGER NOT NULL,
    reset_period reset_period NOT NULL,
    features JSONB NOT NULL DEFAULT '[]',
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Add trigger for updated_at
CREATE TRIGGER set_updated_at
    BEFORE UPDATE ON subscription_plans
    FOR EACH ROW
    EXECUTE FUNCTION trigger_set_timestamp();

-- Add plan_id to empresas table
ALTER TABLE empresas
ADD COLUMN plan_id UUID REFERENCES subscription_plans(id);

-- Insert initial plans
INSERT INTO subscription_plans (name, code, type, price, daily_booking_limit, reset_period, features, is_active)
VALUES 
    ('Plan Gratuito', 'FREE', 'FREE', 0, 90, 'DAILY', '["Reservas básicas", "Gestión de usuarios", "Panel de administración"]', true),
    ('Plan Pro', 'PRO', 'PRO', 49.99, 500, 'DAILY', '["Reservas ilimitadas", "Soporte prioritario", "Análisis avanzados", "Gestión de torneos", "Marketing y promociones"]', true);

-- Update existing empresas to FREE plan
UPDATE empresas
SET plan_id = (SELECT id FROM subscription_plans WHERE code = 'FREE')
WHERE plan_id IS NULL;

Vista Materializada:

CREATE MATERIALIZED VIEW booking_company_view AS
SELECT 
    b.id as booking_id,
    b.date as booking_date,
    b.court_id,
    c.branch_id,
    s.empresa_id,
    e.plan_id
FROM bookings b
JOIN courts c ON b.court_id = c.id
JOIN sedes s ON c.branch_id = s.id
JOIN empresas e ON s.empresa_id = e.id;

-- Crear índices para optimizar el rendimiento
CREATE UNIQUE INDEX idx_booking_company_view_id ON booking_company_view(booking_id);
CREATE INDEX idx_booking_company_view_date ON booking_company_view(booking_date);
CREATE INDEX idx_booking_company_view_empresa ON booking_company_view(empresa_id);
